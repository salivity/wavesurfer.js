Audio Bookmarks with Wavesurfer.js and LocalStorage

This article explains how to build a client-side audio bookmarking system using Wavesurfer.js and the browser's LocalStorage API. You will learn how to initialize an interactive audio waveform, capture specific playback timestamps as user-defined bookmarks, persist those timestamps across browser sessions without a backend database, and render a navigable list to jump directly to saved points in the audio track.

1. HTML and Library Setup

Load Wavesurfer.js via a CDN and set up basic HTML containers for the waveform, playback controls, a bookmark button, and the list where saved timestamps will render.

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Audio Bookmarking</title>
  <!-- Load Wavesurfer.js v7 -->
  <script src="https://unpkg.com/wavesurfer.js@7"></script>
</head>
<body>
  <div id="waveform"></div>
  
  <button id="playPauseBtn">Play</button>
  <button id="bookmarkBtn">Add Bookmark</button>

  <h3>Bookmarks</h3>
  <ul id="bookmarksList"></ul>

  <script src="app.js"></script>
</body>
</html>

2. Initialize Wavesurfer.js

Create an instance of Wavesurfer pointing to the audio file and configure basic playback controls.

// app.js
const STORAGE_KEY = 'audio_track_bookmarks';

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4F4A85',
  progressColor: '#383351',
  url: 'audio.mp3', // Replace with your audio file path
});

const playPauseBtn = document.getElementById('playPauseBtn');
playPauseBtn.addEventListener('click', () => {
  wavesurfer.playPause();
  playPauseBtn.textContent = wavesurfer.isPlaying() ? 'Pause' : 'Play';
});

wavesurfer.on('finish', () => {
  playPauseBtn.textContent = 'Play';
});

3. Manage LocalStorage for Bookmarks

LocalStorage stores data as strings. Define helper functions to retrieve and save arrays of bookmark timestamps.

function getBookmarks() {
  const stored = localStorage.getItem(STORAGE_KEY);
  return stored ? JSON.parse(stored) : [];
}

function saveBookmarks(bookmarks) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(bookmarks));
}

4. Create and Render Bookmarks

Format timestamps into human-readable minutes and seconds, update the DOM, and attach click listeners to seek to the saved time using wavesurfer.setTime().

const bookmarksList = document.getElementById('bookmarksList');
const bookmarkBtn = document.getElementById('bookmarkBtn');

function formatTime(seconds) {
  const mins = Math.floor(seconds / 60);
  const secs = Math.floor(seconds % 60);
  return `${mins}:${secs < 10 ? '0' : ''}${secs}`;
}

function renderBookmarks() {
  bookmarksList.innerHTML = '';
  const bookmarks = getBookmarks();

  bookmarks.sort((a, b) => a - b);

  bookmarks.forEach((time, index) => {
    const li = document.createElement('li');
    
    // Jump to bookmark on click
    const jumpBtn = document.createElement('button');
    jumpBtn.textContent = `Jump to ${formatTime(time)}`;
    jumpBtn.addEventListener('click', () => {
      wavesurfer.setTime(time);
      wavesurfer.play();
      playPauseBtn.textContent = 'Pause';
    });

    // Delete bookmark
    const deleteBtn = document.createElement('button');
    deleteBtn.textContent = '✕';
    deleteBtn.style.marginLeft = '8px';
    deleteBtn.addEventListener('click', () => {
      removeBookmark(index);
    });

    li.appendChild(jumpBtn);
    li.appendChild(deleteBtn);
    bookmarksList.appendChild(li);
  });
}

function addBookmark() {
  const currentTime = wavesurfer.getCurrentTime();
  const bookmarks = getBookmarks();
  
  // Avoid saving duplicate points within the same second
  const exists = bookmarks.some(time => Math.floor(time) === Math.floor(currentTime));
  if (!exists) {
    bookmarks.push(currentTime);
    saveBookmarks(bookmarks);
    renderBookmarks();
  }
}

function removeBookmark(index) {
  const bookmarks = getBookmarks();
  bookmarks.splice(index, 1);
  saveBookmarks(bookmarks);
  renderBookmarks();
}

bookmarkBtn.addEventListener('click', addBookmark);

// Initial render when the script loads
renderBookmarks();

5. Add Waveform Markers (Optional)

If visual markers directly on the waveform are preferred, register the Wavesurfer RegionsPlugin or MarkersPlugin and populate it from the same getBookmarks() array upon initialization. Updating the array in localStorage alongside your visual markers ensures state persistence across page reloads.
