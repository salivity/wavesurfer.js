Render Cropped Audio Segments with Wavesurfer.js

This article explains how to visually render isolated segments of a larger audio file using independent Wavesurfer.js instances. Rather than processing and splitting audio files server-side, you can achieve lightweight, client-side segment visualization by slicing Web Audio API buffers and passing them directly to standalone Wavesurfer containers.

Slicing the AudioBuffer for Standalone Instances

To render a standalone visual segment without loading the full track's waveform, you must slice the original AudioBuffer to contain only the required time range. Passing a trimmed buffer to Wavesurfer ensures that the rendered waveform visually scales from 0% to 100% of the designated container, representing only the chosen start and end points.

1. Fetch and Decode the Audio Source

First, fetch the primary audio asset and decode it into an AudioBuffer using the browser's native AudioContext:

async function getDecodedAudioBuffer(audioUrl) {
  const audioContext = new (window.AudioContext || window.webkitAudioContext)();
  const response = await fetch(audioUrl);
  const arrayBuffer = await response.arrayBuffer();
  return await audioContext.decodeAudioData(arrayBuffer);
}

2. Create the Buffer Slicing Helper

Next, create a helper function that extracts a specific segment from an AudioBuffer based on start and end timestamps (in seconds):

function sliceAudioBuffer(originalBuffer, startTime, endTime) {
  const audioContext = new (window.AudioContext || window.webkitAudioContext)();
  const sampleRate = originalBuffer.sampleRate;
  const numberOfChannels = originalBuffer.numberOfChannels;

  const startOffset = Math.floor(startTime * sampleRate);
  const endOffset = Math.min(Math.floor(endTime * sampleRate), originalBuffer.length);
  const frameCount = endOffset - startOffset;

  if (frameCount <= 0) {
    throw new Error("Invalid start or end boundaries.");
  }

  const croppedBuffer = audioContext.createBuffer(
    numberOfChannels,
    frameCount,
    sampleRate
  );

  for (let channel = 0; channel < numberOfChannels; channel++) {
    const originalChannelData = originalBuffer.getChannelData(channel);
    const croppedChannelData = originalChannelData.subarray(startOffset, endOffset);
    croppedBuffer.copyToChannel(croppedChannelData, channel, 0);
  }

  return croppedBuffer;
}

3. Initialize and Load into Wavesurfer

Once you have the sliced AudioBuffer, create a standalone Wavesurfer instance targeted at a specific DOM element and load the sliced buffer directly.

import WaveSurfer from 'wavesurfer.js';

async function renderSegmentWaveform(containerId, audioUrl, startTime, endTime) {
  const fullBuffer = await getDecodedAudioBuffer(audioUrl);
  const segmentBuffer = sliceAudioBuffer(fullBuffer, startTime, endTime);

  const wavesurfer = WaveSurfer.create({
    container: document.getElementById(containerId),
    waveColor: '#4F46E5',
    progressColor: '#10B981',
    cursorColor: '#374151',
    barWidth: 2,
    barGap: 1,
    barRadius: 2,
    height: 80,
  });

  // Load the cropped AudioBuffer directly
  wavesurfer.loadAudioBuffer(segmentBuffer);

  return wavesurfer;
}

Rendering Multiple Segments on a Single Page

To render multiple cropped snippets from the same master audio track, fetch and decode the original file once, then slice the buffer for each target container:

async function renderMultipleSegments(audioUrl, segments) {
  const fullBuffer = await getDecodedAudioBuffer(audioUrl);

  segments.forEach(({ containerId, start, end }) => {
    const segmentBuffer = sliceAudioBuffer(fullBuffer, start, end);

    const wavesurfer = WaveSurfer.create({
      container: document.getElementById(containerId),
      waveColor: '#9CA3AF',
      progressColor: '#2563EB',
      height: 60,
    });

    wavesurfer.loadAudioBuffer(segmentBuffer);
  });
}

// Example usage:
renderMultipleSegments('path/to/master-track.mp3', [
  { containerId: 'segment-1', start: 10, end: 25 },
  { containerId: 'segment-2', start: 40, end: 55 }
]);

Performance Considerations

  • Cache the Decoded Buffer: Decoding large files using AudioContext.decodeAudioData is CPU-intensive. Decode the master track once in memory, then produce as many cropped instances as needed.
  • Instance Cleanup: Standalone Wavesurfer instances allocate canvas elements and Web Audio nodes. Call wavesurfer.destroy() whenever a rendered segment is removed from the DOM to prevent memory leaks.
