Calculate Real-Time Decibels in Wavesurfer.js

This guide explains how to calculate and display real-time decibel (dB) levels during audio playback in Wavesurfer.js using the Web Audio API's AnalyserNode and getByteFrequencyData. You will learn how the byte frequency scale maps to decibels, how to configure the audio nodes, and how to write a performant rendering loop using requestAnimationFrame.

1. Understanding getByteFrequencyData and Decibels

The Web Audio API's AnalyserNode.getByteFrequencyData() populates a Uint8Array with integers ranging from 0 to 255. These values do not represent raw decibels directly; instead, they represent a linear quantization of the decibel range defined on the AnalyserNode:

  • 0 corresponds to analyser.minDecibels (default: -100 dBFS).
  • 255 corresponds to analyser.maxDecibels (default: -30 dBFS).

To determine the decibel value of any frequency bin, use the following linear interpolation formula:

\[\text{dB} = \text{minDecibels} + \left( \frac{\text{byteValue}}{255} \times (\text{maxDecibels} - \text{minDecibels}) \right)\]

2. Setting Up the AnalyserNode with Wavesurfer.js

Wavesurfer.js plays audio either via the Web Audio API or standard HTML5 <audio> elements. To read frequency data, you must route the audio through an AnalyserNode.

import WaveSurfer from 'wavesurfer.js';

// 1. Initialize Wavesurfer
const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4F46E5',
  progressColor: '#818CF8',
  url: '/path-to-audio.mp3',
});

let analyser;
let dataArray;

// 2. Connect the AnalyserNode once audio is ready
wavesurfer.on('ready', () => {
  const audioContext = wavesurfer.getMediaElement() 
    ? new (window.AudioContext || window.webkitAudioContext)() 
    : wavesurfer.backend.ac;

  analyser = audioContext.createAnalyser();
  analyser.fftSize = 256; // Smaller FFT size = lower latency, fewer bins (128 bins)
  analyser.smoothingTimeConstant = 0.8; // Smooths rapid fluctuations (0.0 to 1.0)
  analyser.minDecibels = -100;
  analyser.maxDecibels = -30;

  // Route media element through the analyser if using HTML5 media element
  const mediaElement = wavesurfer.getMediaElement();
  if (mediaElement) {
    const source = audioContext.createMediaElementSource(mediaElement);
    source.connect(analyser);
    analyser.connect(audioContext.destination);
  }

  // Create the buffer to hold frequency data
  dataArray = new Uint8Array(analyser.frequencyBinCount);
});

3. Calculating Peak and Average Decibels

A decibel meter typically displays either the peak dB (the loudest frequency bin) or the average dB (overall spectral loudness).

function getDecibelLevels(analyser, dataArray) {
  analyser.getByteFrequencyData(dataArray);

  let sum = 0;
  let maxByte = 0;

  for (let i = 0; i < dataArray.length; i++) {
    const value = dataArray[i];
    if (value > maxByte) {
      maxByte = value;
    }
    sum += value;
  }

  const averageByte = sum / dataArray.length;
  const range = analyser.maxDecibels - analyser.minDecibels;

  // Convert raw 0-255 values back to decibels (dBFS)
  const peakDb = analyser.minDecibels + (maxByte / 255) * range;
  const averageDb = analyser.minDecibels + (averageByte / 255) * range;

  return { peakDb, averageDb };
}

4. Running the Animation Loop

To display the meter accurately without blocking the UI, execute the calculation inside a requestAnimationFrame loop that starts when playback begins and cancels when playback pauses.

let animationFrameId = null;

function updateMeter() {
  if (!wavesurfer.isPlaying()) return;

  const { peakDb, averageDb } = getDecibelLevels(analyser, dataArray);

  // Update UI (e.g., text readout or volume bar)
  renderMeterUI(peakDb, averageDb);

  animationFrameId = requestAnimationFrame(updateMeter);
}

// Start loop on play
wavesurfer.on('play', () => {
  cancelAnimationFrame(animationFrameId);
  updateMeter();
});

// Stop loop on pause or finish
wavesurfer.on('pause', () => {
  cancelAnimationFrame(animationFrameId);
});

wavesurfer.on('finish', () => {
  cancelAnimationFrame(animationFrameId);
});

function renderMeterUI(peakDb, averageDb) {
  const meterFill = document.getElementById('meter-fill');
  const meterText = document.getElementById('meter-text');

  // Normalize -100 dBFS to 0 dBFS into a 0% - 100% scale for CSS
  const percentage = Math.max(0, Math.min(100, ((peakDb + 100) / 100) * 100));

  meterFill.style.width = `${percentage}%`;
  meterText.textContent = `${peakDb.toFixed(1)} dB`;
}

5. Best Practices

  • Use fftSize = 128 or 256: Metering does not require high frequency resolution. Smaller buffer sizes reduce memory footprint and calculation overhead.
  • Tune smoothingTimeConstant: A value between 0.7 and 0.85 produces a natural decay characteristic similar to hardware decibel meters.
  • Clamp Visual Values: Decibels in digital systems (dBFS) peak at 0 dB. Ensure values do not exceed 0 dB or drop below your minDecibels floor when rendering visual components.
