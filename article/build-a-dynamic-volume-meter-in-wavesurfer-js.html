Build a Dynamic Volume Meter in Wavesurfer.js

This guide explains how to build a dynamic, real-time volume meter that tracks alongside the Wavesurfer.js playhead. By coupling the Web Audio API’s AnalyserNode with Wavesurfer's playback lifecycle, you will learn how to compute instantaneous Root Mean Square (RMS) audio amplitude and translate it into a visual meter that moves across the waveform synchronously with the playback cursor.

Core Concepts

To display a volume meter directly on the playhead, two parallel systems must work together:

  1. Audio Analysis: An AnalyserNode connected to the Wavesurfer audio source continuously reads frequency/time-domain byte data to calculate decibel levels or RMS amplitude.
  2. Visual Synchronization: An overlay element positioned via CSS tracks the current playback position percentage (currentTime / duration) while scaling its height or color based on the computed volume.

Step 1: Initialize Wavesurfer and Web Audio Nodes

Create your Wavesurfer instance and access its underlying audio context to insert an AnalyserNode.

import WaveSurfer from 'wavesurfer.js';

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4F4A85',
  progressColor: '#383351',
  url: '/path/to/audio.mp3',
});

let analyser;
let dataArray;

wavesurfer.on('ready', () => {
  const audioContext = wavesurfer.getAudioContext();
  const mediaElement = wavesurfer.getMediaElement();

  // Create an AnalyserNode
  analyser = audioContext.createAnalyser();
  analyser.fftSize = 256;

  // Route the media element through the analyser
  const source = audioContext.createMediaElementSource(mediaElement);
  source.connect(analyser);
  analyser.connect(audioContext.destination);

  // Prepare the buffer for reading data
  const bufferLength = analyser.frequencyBinCount;
  dataArray = new Uint8Array(bufferLength);
});

Step 2: Calculate Amplitude (RMS)

Root Mean Square (RMS) provides an accurate representation of perceived loudness. Convert the time-domain byte data (ranging from 0 to 255 with 128 as silence) into a normalized float between 0 and 1.

function getInstantVolume(analyser, dataArray) {
  analyser.getByteTimeDomainData(dataArray);

  let sumSquares = 0;
  for (let i = 0; i < dataArray.length; i++) {
    // Center the 0-255 byte value to a -1.0 to 1.0 range
    const normalized = (dataArray[i] - 128) / 128;
    sumSquares += normalized * normalized;
  }

  // Calculate RMS
  const rms = Math.sqrt(sumSquares / dataArray.length);
  
  // Clamp value between 0 and 1
  return Math.min(Math.max(rms * 2.5, 0), 1);
}

Step 3: Create the HTML and CSS Structure

Place the meter element inside the waveform container. Ensure the container has position: relative so the meter can be placed absolutely along the X-axis.

<div id="waveform-wrapper" style="position: relative; width: 100%;">
  <div id="waveform"></div>
  
  <!-- Volume Meter Container attached to Playhead -->
  <div id="playhead-volume-meter">
    <div id="volume-level-bar"></div>
  </div>
</div>
#playhead-volume-meter {
  position: absolute;
  top: 0;
  bottom: 0;
  width: 4px;
  pointer-events: none;
  transform: translateX(-50%);
  display: flex;
  align-items: flex-end;
  z-index: 10;
}

#volume-level-bar {
  width: 100%;
  background: #00ff88;
  height: 0%;
  transition: height 0.05s ease, background-color 0.05s ease;
  border-radius: 2px;
}

Step 4: Run the Animation Loop

Synchronize the meter's position and scale using requestAnimationFrame. Update the horizontal offset using the current playback percentage and update the bar height using the calculated volume.

const meter = document.getElementById('playhead-volume-meter');
const bar = document.getElementById('volume-level-bar');
let animationFrameId;

function updateMeter() {
  if (wavesurfer.isPlaying()) {
    // 1. Update playhead position
    const currentTime = wavesurfer.getCurrentTime();
    const duration = wavesurfer.getDuration();
    const progress = (currentTime / duration) * 100;
    
    meter.style.left = `${progress}%`;

    // 2. Update volume height
    const volume = getInstantVolume(analyser, dataArray);
    const heightPercent = volume * 100;
    bar.style.height = `${heightPercent}%`;

    // Optional: Dynamic color thresholding
    if (volume > 0.8) {
      bar.style.backgroundColor = '#ff4d4d'; // Peaking / Red
    } else if (volume > 0.5) {
      bar.style.backgroundColor = '#ffaa00'; // Warning / Orange
    } else {
      bar.style.backgroundColor = '#00ff88'; // Normal / Green
    }

    animationFrameId = requestAnimationFrame(updateMeter);
  }
}

// Attach lifecycle events
wavesurfer.on('play', () => {
  cancelAnimationFrame(animationFrameId);
  animationFrameId = requestAnimationFrame(updateMeter);
});

wavesurfer.on('pause', () => {
  cancelAnimationFrame(animationFrameId);
  bar.style.height = '0%';
});

wavesurfer.on('seeking', (position) => {
  meter.style.left = `${position * 100}%`;
  bar.style.height = '0%';
});

wavesurfer.on('finish', () => {
  cancelAnimationFrame(animationFrameId);
  bar.style.height = '0%';
});

Key Considerations

  • CORS Restrictions: When loading audio from an external domain, ensure the server sets appropriate Access-Control-Allow-Origin headers. Without CORS headers, the Web Audio API will output silence.
  • AudioContext Suspension: Browsers often initialize AudioContext in a suspended state. Wavesurfer resumes this context automatically on user interaction, but ensure your audio routing executes after the first play action if initialized manually.
  • Performance: Running getByteTimeDomainData inside requestAnimationFrame is lightweight for typical FFT sizes (128 to 256), maintaining a smooth 60 FPS update rate alongside playback.
