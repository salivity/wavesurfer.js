Bind Wavesurfer.js Controls in Angular

Integrating Wavesurfer.js into an Angular component requires referencing a DOM container using Angular's ElementRef, initializing the audio instance during the appropriate lifecycle hook, and binding custom UI controls to component methods. This guide demonstrates how to install Wavesurfer.js, render an interactive waveform, map playback controls like play, pause, and volume using Angular event bindings, and clean up audio instances to prevent memory leaks.

1. Install Wavesurfer.js

First, install the Wavesurfer.js package in your Angular project via npm:

npm install wavesurfer.js

2. Define the Component Template

In your component template (audio-player.component.html), create a container element for the waveform using an Angular template reference variable (#waveform). Add your control buttons and inputs using Angular's standard event binding syntax (click) and (input).

<div class="player-container">
  <!-- Waveform Container -->
  <div #waveform id="waveform"></div>

  <!-- Player Controls -->
  <div class="controls">
    <button (click)="playPause()">{{ isPlaying ? 'Pause' : 'Play' }}</button>
    <button (click)="stop()">Stop</button>
    <button (click)="toggleMute()">{{ isMuted ? 'Unmute' : 'Mute' }}</button>

    <label for="volume">Volume</label>
    <input
      id="volume"
      type="range"
      min="0"
      max="1"
      step="0.05"
      [value]="volume"
      (input)="onVolumeChange($event)"
    />
  </div>
</div>

3. Implement the Component Logic

In your TypeScript file (audio-player.component.ts), import WaveSurfer and bind it to the view. Use the AfterViewInit lifecycle hook to ensure the waveform container element is available in the DOM before initializing the player.

import { Component, ElementRef, ViewChild, AfterViewInit, OnDestroy } from '@angular/core';
import WaveSurfer from 'wavesurfer.js';

@Component({
  selector: 'app-audio-player',
  templateUrl: './audio-player.component.html',
  styleUrls: ['./audio-player.component.css']
})
export class AudioPlayerComponent implements AfterViewInit, OnDestroy {
  @ViewChild('waveform', { static: false }) waveformRef!: ElementRef<HTMLDivElement>;

  private waveSurfer!: WaveSurfer;
  public isPlaying: boolean = false;
  public isMuted: boolean = false;
  public volume: number = 0.8;

  ngAfterViewInit(): void {
    this.initWaveSurfer();
  }

  private initWaveSurfer(): void {
    this.waveSurfer = WaveSurfer.create({
      container: this.waveformRef.nativeElement,
      waveColor: '#4F4A85',
      progressColor: '#383351',
      url: 'assets/sample-audio.mp3', // Replace with your audio source
      height: 100,
      barWidth: 2,
      barGap: 1
    });

    // Listen to Wavesurfer events to synchronize component state
    this.waveSurfer.on('play', () => (this.isPlaying = true));
    this.waveSurfer.on('pause', () => (this.isPlaying = false));
    this.waveSurfer.on('finish', () => (this.isPlaying = false));
  }

  public playPause(): void {
    this.waveSurfer.playPause();
  }

  public stop(): void {
    this.waveSurfer.stop();
    this.isPlaying = false;
  }

  public toggleMute(): void {
    this.isMuted = !this.isMuted;
    this.waveSurfer.setMuted(this.isMuted);
  }

  public onVolumeChange(event: Event): void {
    const target = event.target as HTMLInputElement;
    this.volume = parseFloat(target.value);
    this.waveSurfer.setVolume(this.volume);
    if (this.isMuted && this.volume > 0) {
      this.isMuted = false;
    }
  }

  ngOnDestroy(): void {
    if (this.waveSurfer) {
      this.waveSurfer.destroy();
    }
  }
}

Key Considerations

  • Lifecycle Timing: Always initialize WaveSurfer.create() inside ngAfterViewInit to guarantee that the ElementRef points to a rendered DOM node.
  • State Synchronization: Listen to native Wavesurfer events like 'play', 'pause', and 'finish' to keep Angular component variables aligned with the audio playback state.
  • Resource Cleanup: Always call this.waveSurfer.destroy() inside the ngOnDestroy hook to remove event listeners and free browser audio resources when the component is unmounted.
