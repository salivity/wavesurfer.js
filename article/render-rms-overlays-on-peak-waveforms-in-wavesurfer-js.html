Render RMS Overlays on Peak Waveforms in Wavesurfer.js

This guide explains how to calculate Root Mean Square (RMS) values from an audio file and render them directly over standard peak waveforms using Wavesurfer.js. By overlaying RMS data onto peak waveforms, you can provide users with a visual representation of both transient dynamic peaks and perceived loudness simultaneously. The following sections walk through extracting audio data, computing RMS alongside peak values, and drawing the overlay using HTML5 Canvas.

Understanding Peaks vs. RMS

Peak waveforms represent the absolute highest amplitude reached within discrete time slices, making them ideal for spotting clipping and transient spikes. Root Mean Square (RMS) measurements calculate the continuous signal power over time:

\[\text{RMS} = \sqrt{\frac{1}{N} \sum_{i=1}^{N} x_i^2}\]

Overlaying an RMS trace inside a standard peak waveform offers a dual-layer visualization where the outer boundary displays dynamic peaks and an inner, darker or distinctively colored core displays perceived acoustic energy.

Step 1: Initialize Wavesurfer.js

Set up your container and create the base Wavesurfer instance. Ensure the container has relative positioning so the secondary RMS overlay can be aligned precisely on top of the default waveform.

<div id="waveform-container" style="position: relative; width: 100%; height: 128px;">
  <div id="waveform" style="width: 100%; height: 100%;"></div>
  <canvas id="rms-overlay" style="position: absolute; top: 0; left: 0; width: 100%; height: 100%; pointer-events: none;"></canvas>
</div>
import WaveSurfer from 'wavesurfer.js';

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4f46e5',
  progressColor: '#312e81',
  cursorColor: '#ffffff',
  height: 128,
  barWidth: 2,
  barGap: 1,
  barRadius: 2,
});

Step 2: Compute Peak and RMS Data

Fetch and decode the audio file using the Web Audio API to access raw PCM channel data. Then, break the samples into windows matching the total number of bars to calculate both peak amplitude and RMS energy for each window.

async function extractAudioData(audioUrl, numberOfBars) {
  const response = await fetch(audioUrl);
  const arrayBuffer = await response.arrayBuffer();
  const audioContext = new (window.AudioContext || window.webkitAudioContext)();
  const audioBuffer = await audioContext.decodeAudioData(arrayBuffer);

  const channelData = audioBuffer.getChannelData(0); // Left channel or mono
  const totalSamples = channelData.length;
  const blockSize = Math.floor(totalSamples / numberOfBars);

  const peaks = [];
  const rmsValues = [];

  for (let i = 0; i < numberOfBars; i++) {
    const start = i * blockSize;
    const end = start + blockSize;
    let max = 0;
    let sumSquares = 0;

    for (let j = start; j < end; j++) {
      const sample = channelData[j];
      const absSample = Math.abs(sample);
      if (absSample > max) {
        max = absSample;
      }
      sumSquares += sample * sample;
    }

    peaks.push(max);
    rmsValues.push(Math.sqrt(sumSquares / blockSize));
  }

  return { peaks, rmsValues, audioBuffer };
}

Step 3: Draw the RMS Overlay

Match the overlay canvas dimensions to the Wavesurfer canvas. Iterate through the precomputed RMS values to draw bars centered along the horizontal axis, nested directly within the peaks rendered by Wavesurfer.

function drawRmsOverlay(canvas, rmsValues, barWidth, barGap, barRadius, color) {
  const ctx = canvas.getContext('2d');
  const dpr = window.devicePixelRatio || 1;
  const width = canvas.clientWidth;
  const height = canvas.clientHeight;

  canvas.width = width * dpr;
  canvas.height = height * dpr;
  ctx.scale(dpr, dpr);
  ctx.clearRect(0, 0, width, height);

  ctx.fillStyle = color;
  const centerY = height / 2;
  const step = barWidth + barGap;

  rmsValues.forEach((rms, index) => {
    const x = index * step;
    // Scale the RMS height relative to total canvas height
    const barHeight = Math.max(rms * height, 2);
    const y = centerY - barHeight / 2;

    if (barRadius > 0 && ctx.roundRect) {
      ctx.beginPath();
      ctx.roundRect(x, y, barWidth, barHeight, barRadius);
      ctx.fill();
    } else {
      ctx.fillRect(x, y, barWidth, barHeight);
    }
  });
}

Step 4: Putting It Together

Calculate the number of bars dynamically based on container width, run the data extraction, load the audio into Wavesurfer using the precomputed peaks, and render the RMS overlay.

async function initWaveformWithRms(audioUrl) {
  const container = document.getElementById('waveform-container');
  const overlayCanvas = document.getElementById('rms-overlay');
  
  const barWidth = 2;
  const barGap = 1;
  const barRadius = 2;
  const totalWidth = container.clientWidth;
  const numberOfBars = Math.floor(totalWidth / (barWidth + barGap));

  const { peaks, rmsValues } = await extractAudioData(audioUrl, numberOfBars);

  // Load into Wavesurfer with custom peak values
  wavesurfer.load(audioUrl, [peaks]);

  wavesurfer.on('ready', () => {
    // Render the overlay using a higher-contrast color inside the peak bars
    drawRmsOverlay(overlayCanvas, rmsValues, barWidth, barGap, barRadius, 'rgba(255, 255, 255, 0.7)');
  });

  // Re-render when resizing the window
  window.addEventListener('resize', async () => {
    const newWidth = container.clientWidth;
    const newBarCount = Math.floor(newWidth / (barWidth + barGap));
    const updatedData = await extractAudioData(audioUrl, newBarCount);
    drawRmsOverlay(overlayCanvas, updatedData.rmsValues, barWidth, barGap, barRadius, 'rgba(255, 255, 255, 0.7)');
  });
}

initWaveformWithRms('path/to/audio.mp3');

Step 5: Handling Playback Progress

To match Wavesurfer's progress coloration on the RMS layer, use a canvas clipping path or apply composite operations in the overlay drawing function. Alternatively, split the overlay into two canvases—one for the unplayed state and one for the played state—and update the width or clip-path of the played canvas during the Wavesurfer timeupdate event.
