Build an Audio Trimmer with Wavesurfer.js

This guide explains how to construct an online audio trimming tool using Wavesurfer.js and standard browser APIs. You will learn how to initialize the waveform visualization, enable selection handles with the Regions plugin, extract the selected audio segments using the Web Audio API, and export the trimmed result into a downloadable file.

1. Set Up the Project Structure

Create an HTML file with basic markup to hold your waveform, playback controls, file input, and export button. Load Wavesurfer.js and the Regions plugin via CDN or install them via npm.

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>Audio Trimmer</title>
  <script src="https://unpkg.com/wavesurfer.js@7"></script>
  <script src="https://unpkg.com/wavesurfer.js@7/dist/plugins/regions.min.js"></script>
</head>
<body>
  <input type="file" id="audio-input" accept="audio/*" />
  <div id="waveform" style="width: 100%; margin: 20px 0;"></div>
  
  <button id="btn-play">Play / Pause</button>
  <button id="btn-trim">Trim and Download</button>

  <script src="app.js"></script>
</body>
</html>

2. Initialize Wavesurfer and the Regions Plugin

In your JavaScript file (app.js), instantiate Wavesurfer alongside the Regions plugin. The Regions plugin allows users to visually define the start and end points of the cut.

const regions = WaveSurfer.Regions.create();

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4F4A85',
  progressColor: '#383351',
  plugins: [regions],
});

let activeRegion = null;

// Handle audio upload
document.getElementById('audio-input').addEventListener('change', (e) => {
  const file = e.target.files[0];
  if (file) {
    const url = URL.createObjectURL(file);
    wavesurfer.load(url);
  }
});

// Controls
document.getElementById('btn-play').addEventListener('click', () => {
  wavesurfer.playPause();
});

3. Create a Trimming Selection Region

Once the audio loads, clear any previous regions and add a single region that the user can resize and drag. Restrict user interaction to one region to keep the trimming workflow simple.

wavesurfer.on('ready', () => {
  regions.clearRegions();
  
  const duration = wavesurfer.getDuration();
  
  // Create an initial region covering a portion of the audio
  activeRegion = regions.addRegion({
    start: duration * 0.1,
    end: duration * 0.9,
    color: 'rgba(0, 255, 0, 0.2)',
    drag: true,
    resize: true,
  });
});

// Update the active reference if the region changes
regions.on('region-updated', (region) => {
  activeRegion = region;
});

4. Slice the AudioBuffer

To trim the audio, obtain the decoded AudioBuffer from Wavesurfer, calculate the sample frames corresponding to the region’s start and end times, and copy those samples into a new AudioBuffer.

function sliceAudioBuffer(originalBuffer, start, end) {
  const sampleRate = originalBuffer.sampleRate;
  const numberOfChannels = originalBuffer.numberOfChannels;
  
  const startOffset = Math.floor(start * sampleRate);
  const endOffset = Math.floor(end * sampleRate);
  const frameCount = endOffset - startOffset;

  const audioContext = new (window.AudioContext || window.webkitAudioContext)();
  const trimmedBuffer = audioContext.createBuffer(numberOfChannels, frameCount, sampleRate);

  for (let channel = 0; channel < numberOfChannels; channel++) {
    const originalChannelData = originalBuffer.getChannelData(channel);
    const trimmedChannelData = trimmedBuffer.getChannelData(channel);
    for (let i = 0; i < frameCount; i++) {
      trimmedChannelData[i] = originalChannelData[startOffset + i];
    }
  }

  return trimmedBuffer;
}

5. Encode and Export the Audio to WAV

The sliced AudioBuffer must be converted into a binary format such as WAV to be saved to the user's device. Implement a WAV encoder function to write the buffer's PCM data to a Blob.

function audioBufferToWav(buffer) {
  const numChannels = buffer.numberOfChannels;
  const sampleRate = buffer.sampleRate;
  const format = 1; // PCM
  const bitDepth = 16;
  
  const bytesPerSample = bitDepth / 8;
  const blockAlign = numChannels * bytesPerSample;
  
  const numSamples = buffer.length;
  const byteRate = sampleRate * blockAlign;
  const dataSize = numSamples * blockAlign;
  const bufferSize = 44 + dataSize;
  
  const arrayBuffer = new ArrayBuffer(bufferSize);
  const view = new DataView(arrayBuffer);

  function writeString(offset, string) {
    for (let i = 0; i < string.length; i++) {
      view.setUint8(offset + i, string.charCodeAt(i));
    }
  }

  // RIFF Chunk Descriptor
  writeString(0, 'RIFF');
  view.setUint32(4, 36 + dataSize, true);
  writeString(8, 'WAVE');
  
  // "fmt " sub-chunk
  writeString(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, format, true);
  view.setUint16(22, numChannels, true);
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, byteRate, true);
  view.setUint16(32, blockAlign, true);
  view.setUint16(34, bitDepth, true);
  
  // "data" sub-chunk
  writeString(36, 'data');
  view.setUint32(40, dataSize, true);

  // Write interleaved PCM samples
  let offset = 44;
  for (let i = 0; i < numSamples; i++) {
    for (let channel = 0; channel < numChannels; channel++) {
      const sample = Math.max(-1, Math.min(1, buffer.getChannelData(channel)[i]));
      view.setInt16(offset, sample < 0 ? sample * 0x8000 : sample * 0x7FFF, true);
      offset += 2;
    }
  }

  return new Blob([view], { type: 'audio/wav' });
}

6. Connect the Trim Event

Attach a click handler to the trim button. It retrieves the current decoded audio buffer from Wavesurfer, processes the slice using the region bounds, creates a file download link, and clicks it automatically.

document.getElementById('btn-trim').addEventListener('click', () => {
  if (!activeRegion) return;

  const originalBuffer = wavesurfer.getDecodedData();
  if (!originalBuffer) return;

  const trimmedBuffer = sliceAudioBuffer(
    originalBuffer,
    activeRegion.start,
    activeRegion.end
  );

  const wavBlob = audioBufferToWav(trimmedBuffer);
  const downloadUrl = URL.createObjectURL(wavBlob);

  const downloadLink = document.createElement('a');
  downloadLink.href = downloadUrl;
  downloadLink.download = 'trimmed-audio.wav';
  downloadLink.click();

  URL.revokeObjectURL(downloadUrl);
});
