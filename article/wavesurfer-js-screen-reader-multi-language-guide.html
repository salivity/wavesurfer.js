Wavesurfer.js Screen Reader Multi-Language Guide

Implementing multi-language accessibility for Wavesurfer.js controls requires pairing semantic HTML elements with localized Accessible Rich Internet Applications (ARIA) attributes and dynamic state management. Because Wavesurfer.js primarily renders waveforms using the HTML <canvas> element—which is inherently inaccessible to screen readers—developers must build an external control interface and live status regions. This guide details how to configure a localization schema, bind translated ARIA attributes to player controls, and provide localized live screen reader announcements across multiple languages.

1. Define a Localization Schema

Create a translation dictionary that maps UI actions and dynamic playback states to specific language codes. This ensures all control labels and status updates can be retrieved dynamically based on the user's active locale.

const i18n = {
  en: {
    play: "Play audio",
    pause: "Pause audio",
    seek: "Audio timeline",
    volume: "Volume",
    mute: "Mute audio",
    unmute: "Unmute audio",
    statusPlaying: "Audio playing",
    statusPaused: "Audio paused at {time}",
  },
  es: {
    play: "Reproducir audio",
    pause: "Pausar audio",
    seek: "Línea de tiempo de audio",
    volume: "Volumen",
    mute: "Silenciar audio",
    unmute: "Activar sonido",
    statusPlaying: "Reproduciendo audio",
    statusPaused: "Audio pausado en {time}",
  },
  fr: {
    play: "Lire l'audio",
    pause: "Mettre en pause",
    seek: "Piste audio",
    volume: "Volume",
    mute: "Couper le son",
    unmute: "Activer le son",
    statusPlaying: "Lecture en cours",
    statusPaused: "Audio en pause à {time}",
  }
};

let currentLocale = 'en';

2. Create Semantic HTML with ARIA Attributes

Build custom controls outside the Wavesurfer canvas wrapper. Use native HTML <button> and <input type="range"> elements to ensure built-in keyboard accessibility, and add an aria-live container to convey dynamic state changes to assistive technologies.

<div id="player-container" role="region" aria-label="Audio Player">
  <!-- Dynamic live region for screen reader notifications -->
  <div id="sr-announcements" class="sr-only" aria-live="polite" aria-atomic="true"></div>

  <!-- Player Controls -->
  <button id="play-btn" type="button" aria-label="Play audio"></button>
  <button id="mute-btn" type="button" aria-label="Mute audio"></button>
  
  <input 
    id="seek-slider" 
    type="range" 
    min="0" 
    max="100" 
    value="0" 
    aria-label="Audio timeline"
  />

  <div id="waveform" aria-hidden="true"></div>
</div>

Ensure the visual canvas generated by Wavesurfer.js uses aria-hidden="true" so that screen readers bypass the canvas and interact exclusively with the accessible controls.

3. Synchronize Wavesurfer Events with Localized Labels

Bind event listeners to Wavesurfer.js lifecycle hooks to dynamically update the aria-label attributes as states change (such as toggling between play and pause).

import WaveSurfer from 'wavesurfer.js';

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4F4A85',
  progressColor: '#383351',
  url: '/path/to/audio.mp3',
});

const playButton = document.getElementById('play-btn');
const srAnnouncements = document.getElementById('sr-announcements');

function updatePlayButtonAria() {
  const isPlaying = wavesurfer.isPlaying();
  const label = isPlaying ? i18n[currentLocale].pause : i18n[currentLocale].play;
  
  playButton.setAttribute('aria-label', label);
  playButton.textContent = label;
}

wavesurfer.on('play', () => {
  updatePlayButtonAria();
  announceToScreenReader(i18n[currentLocale].statusPlaying);
});

wavesurfer.on('pause', () => {
  updatePlayButtonAria();
  const currentTime = formatTime(wavesurfer.getCurrentTime());
  const message = i18n[currentLocale].statusPaused.replace('{time}', currentTime);
  announceToScreenReader(message);
});

function announceToScreenReader(message) {
  srAnnouncements.textContent = '';
  // Small timeout ensures screen readers register the text mutation
  setTimeout(() => {
    srAnnouncements.textContent = message;
  }, 50);
}

function formatTime(seconds) {
  const mins = Math.floor(seconds / 60);
  const secs = Math.floor(seconds % 60);
  return `${mins}:${secs < 10 ? '0' : ''}${secs}`;
}

4. Implement Dynamic Language Switching

When a user switches languages, the interface must update the HTML lang attribute on the player container and re-render all control labels immediately.

function setLanguage(locale) {
  if (!i18n[locale]) return;
  
  currentLocale = locale;
  
  // Set the language context for assistive technologies
  const playerContainer = document.getElementById('player-container');
  playerContainer.setAttribute('lang', locale);

  // Update control labels
  updatePlayButtonAria();
  
  const isMuted = wavesurfer.getMuted();
  const muteLabel = isMuted ? i18n[currentLocale].unmute : i18n[currentLocale].mute;
  document.getElementById('mute-btn').setAttribute('aria-label', muteLabel);

  document.getElementById('seek-slider').setAttribute('aria-label', i18n[currentLocale].seek);
}

// Example: Switch to Spanish
document.getElementById('lang-select').addEventListener('change', (e) => {
  setLanguage(e.target.value);
});

5. Managing Keyboard-Based Seek Adjustments

Ensure that manual progress changes trigger localized accessibility updates. When users modify the seek range with arrow keys, announce the current location in the selected language.

const seekSlider = document.getElementById('seek-slider');

seekSlider.addEventListener('input', (e) => {
  const progress = e.target.value / 100;
  wavesurfer.seekTo(progress);
  
  const currentTime = formatTime(wavesurfer.getDuration() * progress);
  seekSlider.setAttribute('aria-valuetext', currentTime);
});

wavesurfer.on('timeupdate', (currentTime) => {
  const duration = wavesurfer.getDuration();
  if (duration) {
    const value = (currentTime / duration) * 100;
    seekSlider.value = value;
    seekSlider.setAttribute('aria-valuetext', formatTime(currentTime));
  }
});

Using standard aria-valuetext populated with localized time formatting allows screen readers to natively vocalize position changes accurately according to the active language preferences.
