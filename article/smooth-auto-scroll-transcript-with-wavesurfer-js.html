Smooth Auto-Scroll Transcript with Wavesurfer.js

Synchronizing an interactive transcript with Wavesurfer.js requires continuously tracking the playhead, identifying the corresponding dialogue segment, and shifting the transcript viewport smoothly without causing layout thrashing or interfering with manual user navigation. This guide outlines the exact logic and implementation needed to create a performant, jitter-free auto-scrolling transcript using Wavesurfer.js playback events, state comparison, and native smooth scrolling APIs.

1. Listen to Playback Updates

Wavesurfer.js provides the timeupdate event (or audioprocess in older versions), which fires continuously as the audio plays. Instead of triggering a scroll event on every tick, you must query the current playback timestamp and locate the corresponding transcript segment.

wavesurfer.on('timeupdate', (currentTime) => {
  updateTranscript(currentTime);
});

2. Identify the Active Transcript Segment

Each segment in your transcript data must have defined start and end timestamps along with a unique identifier or DOM reference. Use array iteration or a binary search to find the active cue based on currentTime.

const activeCue = transcriptData.find(
  (cue) => currentTime >= cue.start && currentTime < cue.end
);

3. Scroll Only on Active Segment Changes

The primary cause of stuttering and performance issues in auto-scrolling transcripts is calling the scroll API continuously on every timeupdate tick. You must store the currently active cue's ID in memory and only trigger a scroll operation when the active cue ID changes.

let currentActiveId = null;

function updateTranscript(currentTime) {
  const activeCue = transcriptData.find(
    (cue) => currentTime >= cue.start && currentTime < cue.end
  );

  if (!activeCue || activeCue.id === currentActiveId) {
    return;
  }

  currentActiveId = activeCue.id;
  highlightAndScroll(activeCue.id);
}

4. Apply Smooth Scrolling Logic

The cleanest method to scroll the transcript container is using Element.scrollIntoView() configured with behavior: 'smooth' and block: 'center'. Centering the active cue keeps upcoming text visible and provides a stable reading frame.

function highlightAndScroll(cueId) {
  // Update visual classes
  document.querySelectorAll('.transcript-line.active').forEach((el) => {
    el.classList.remove('active');
  });

  const activeElement = document.getElementById(`cue-${cueId}`);
  if (activeElement) {
    activeElement.classList.add('active');

    // Trigger smooth scroll
    if (isAutoScrollEnabled) {
      activeElement.scrollIntoView({
        behavior: 'smooth',
        block: 'center',
        inline: 'nearest'
      });
    }
  }
}

5. Handle User Scroll Interrupts

For a natural user experience, auto-scrolling must temporarily pause when a user manually scrolls the transcript container to read ahead or review previous lines.

Attach event listeners to the transcript container to detect manual interaction:

let isAutoScrollEnabled = true;
let userScrollTimeout = null;
const container = document.getElementById('transcript-container');

container.addEventListener('wheel', handleManualScroll);
container.addEventListener('touchmove', handleManualScroll);

function handleManualScroll() {
  isAutoScrollEnabled = false;

  // Clear existing timeout
  clearTimeout(userScrollTimeout);

  // Resume auto-scroll after 3 seconds of inactivity
  userScrollTimeout = setTimeout(() => {
    isAutoScrollEnabled = true;
  }, 3000);
}

Alternatively, you can provide an explicit toggle button in the UI allowing the user to disable or enable "Follow Audio" at will.

Complete Implementation Example

class TranscriptSync {
  constructor(wavesurfer, transcriptData, containerElement) {
    this.wavesurfer = wavesurfer;
    this.data = transcriptData;
    this.container = containerElement;
    this.currentActiveId = null;
    this.autoScroll = true;
    this.timeout = null;

    this.init();
  }

  init() {
    this.wavesurfer.on('timeupdate', (time) => this.onTimeUpdate(time));

    const disableScroll = () => {
      this.autoScroll = false;
      clearTimeout(this.timeout);
      this.timeout = setTimeout(() => {
        this.autoScroll = true;
      }, 3500);
    };

    this.container.addEventListener('wheel', disableScroll, { passive: true });
    this.container.addEventListener('touchmove', disableScroll, { passive: true });
  }

  onTimeUpdate(currentTime) {
    const cue = this.data.find(
      (item) => currentTime >= item.start && currentTime < item.end
    );

    if (!cue || cue.id === this.currentActiveId) return;

    this.currentActiveId = cue.id;
    this.updateUI(cue.id);
  }

  updateUI(id) {
    const prev = this.container.querySelector('.active');
    if (prev) prev.classList.remove('active');

    const next = document.getElementById(`cue-${id}`);
    if (next) {
      next.classList.add('active');
      if (this.autoScroll) {
        next.scrollIntoView({
          behavior: 'smooth',
          block: 'center'
        });
      }
    }
  }
}
