Sync MediaSession Metadata with Wavesurfer.js

This guide explains how to integrate the browser's native Media Session API with a Wavesurfer.js audio player. By synchronizing Wavesurfer's playback events and audio methods with the navigator.mediaSession interface, you can display rich track metadata (such as titles, artists, and album artwork) on operating system lock screens, notification trays, and wearable devices, while also enabling hardware media keys to control your audio waveform directly.

1. Setting Up Media Metadata

Before binding controls, verify browser support for the Media Session API and define a MediaMetadata object containing the track details.

import WaveSurfer from 'wavesurfer.js';

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4F4A85',
  progressColor: '#383351',
  url: '/audio/track.mp3',
});

function updateMediaSessionMetadata(track) {
  if ('mediaSession' in navigator) {
    navigator.mediaSession.metadata = new MediaMetadata({
      title: track.title,
      artist: track.artist,
      album: track.album,
      artwork: [
        { src: track.artworkSmall, sizes: '96x96', type: 'image/png' },
        { src: track.artworkLarge, sizes: '512x512', type: 'image/png' },
      ],
    });
  }
}

// Example invocation
updateMediaSessionMetadata({
  title: 'Synthwave Sunset',
  artist: 'Retro Horizon',
  album: 'Neon Dreams',
  artworkSmall: '/images/album-96.png',
  artworkLarge: '/images/album-512.png',
});

2. Synchronizing Playback State

Wavesurfer emits events when its state changes. Use these events to keep the operating system's playback state (playing, paused, or none) in sync.

if ('mediaSession' in navigator) {
  wavesurfer.on('play', () => {
    navigator.mediaSession.playbackState = 'playing';
  });

  wavesurfer.on('pause', () => {
    navigator.mediaSession.playbackState = 'paused';
  });

  wavesurfer.on('finish', () => {
    navigator.mediaSession.playbackState = 'none';
  });
}

3. Binding Action Handlers

Action handlers allow hardware media keys and notification controls to operate your Wavesurfer instance. Use navigator.mediaSession.setActionHandler to capture actions such as play, pause, seek, and track skips.

if ('mediaSession' in navigator) {
  // Play and Pause
  navigator.mediaSession.setActionHandler('play', () => {
    wavesurfer.play();
  });

  navigator.mediaSession.setActionHandler('pause', () => {
    wavesurfer.pause();
  });

  // Seeking forward and backward
  const seekOffset = 10; // seconds

  navigator.mediaSession.setActionHandler('seekbackward', () => {
    const newTime = Math.max(wavesurfer.getCurrentTime() - seekOffset, 0);
    wavesurfer.setTime(newTime);
  });

  navigator.mediaSession.setActionHandler('seekforward', () => {
    const newTime = Math.min(wavesurfer.getCurrentTime() + seekOffset, wavesurfer.getDuration());
    wavesurfer.setTime(newTime);
  });

  // Native timeline scrubber support
  navigator.mediaSession.setActionHandler('seekto', (details) => {
    if (details.seekTime !== undefined) {
      wavesurfer.setTime(details.seekTime);
    }
  });

  // Playlist navigation (optional)
  navigator.mediaSession.setActionHandler('previoustrack', () => {
    // Implement custom playlist previous track logic
  });

  navigator.mediaSession.setActionHandler('nexttrack', () => {
    // Implement custom playlist next track logic
  });
}

4. Updating the Scrubber Position State

To keep system timelines, lock screen scrubbers, and smartwatches accurate during playback, call setPositionState using Wavesurfer's timing events.

function updatePositionState() {
  if ('mediaSession' in navigator && 'setPositionState' in navigator.mediaSession) {
    const duration = wavesurfer.getDuration();
    if (duration && !isNaN(duration)) {
      navigator.mediaSession.setPositionState({
        duration: duration,
        playbackRate: wavesurfer.getPlaybackRate(),
        position: wavesurfer.getCurrentTime(),
      });
    }
  }
}

// Update scrubber on waveform ready, seeking, and during playback
wavesurfer.on('ready', updatePositionState);
wavesurfer.on('seeking', updatePositionState);
wavesurfer.on('timeupdate', updatePositionState);

5. Cleaning Up on Component Unmount

When destroying the Wavesurfer instance or unmounting your player component, reset the Media Session handlers to avoid memory leaks or orphaned controls.

function cleanupMediaSession() {
  if ('mediaSession' in navigator) {
    navigator.mediaSession.metadata = null;
    navigator.mediaSession.playbackState = 'none';

    const actions = ['play', 'pause', 'seekbackward', 'seekforward', 'seekto', 'previoustrack', 'nexttrack'];
    actions.forEach(action => {
      try {
        navigator.mediaSession.setActionHandler(action, null);
      } catch (error) {
        // Safe fallback for unsupported action types
      }
    });
  }
}

wavesurfer.on('destroy', cleanupMediaSession);
