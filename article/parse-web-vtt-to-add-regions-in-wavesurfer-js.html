Parse WebVTT to Add Regions in Wavesurfer.js

This article explains how to parse WebVTT caption files and dynamically populate them as interactive audio segments using the Wavesurfer.js Regions plugin. By reading timecodes and cue text from standard .vtt data, you can automatically generate synchronized, visually distinct regions on your waveform for transcription, audio editing, or navigation.

1. Initialize Wavesurfer and the Regions Plugin

Start by setting up the Wavesurfer instance alongside the Regions plugin. In modern Wavesurfer.js (v7+), plugins are registered during initialization using the plugins array.

import WaveSurfer from 'wavesurfer.js';
import RegionsPlugin from 'wavesurfer.js/dist/plugins/regions.esm.js';

const wsRegions = RegionsPlugin.create();

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#d9dcff',
  progressColor: '#4353ff',
  url: '/path/to/audio.mp3',
  plugins: [wsRegions],
});

2. Create the WebVTT Parser

WebVTT files represent time intervals in either HH:MM:SS.mmm or MM:SS.mmm format, separated by -->. To map these cues to Wavesurfer regions, parse the timestamps into total seconds and extract the associated text.

function parseVTTTime(timeString) {
  const parts = timeString.trim().split(':');
  let seconds = 0;

  if (parts.length === 3) {
    seconds += parseFloat(parts[0]) * 3600;
    seconds += parseFloat(parts[1]) * 60;
    seconds += parseFloat(parts[2]);
  } else if (parts.length === 2) {
    seconds += parseFloat(parts[0]) * 60;
    seconds += parseFloat(parts[1]);
  }

  return seconds;
}

function parseWebVTT(vttContent) {
  const cueRegex = /(?:(\d{2}:)?\d{2}:\d{2}\.\d{3})\s*-->\s*(?:(\d{2}:)?\d{2}:\d{2}\.\d{3})/g;
  const lines = vttContent.replace(/\r\n|\r/g, '\n').split('\n');
  const cues = [];

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    
    if (line.includes('-->')) {
      const times = line.split('-->');
      const start = parseVTTTime(times[0]);
      const end = parseVTTTime(times[1]);
      
      let text = '';
      i++;
      while (i < lines.length && lines[i].trim() !== '') {
        text += (text ? '\n' : '') + lines[i].trim();
        i++;
      }

      cues.push({ start, end, text });
    }
  }

  return cues;
}

3. Fetch and Populate Regions

Fetch the WebVTT file, parse its raw text, and iterate through the extracted cues. Use the addRegion method on the Regions plugin instance to inject each cue directly onto the waveform.

async function loadVTTRegions(vttUrl) {
  const response = await fetch(vttUrl);
  const vttText = await response.text();
  const cues = parseWebVTT(vttText);

  cues.forEach((cue) => {
    wsRegions.addRegion({
      start: cue.start,
      end: cue.end,
      content: cue.text,
      color: 'rgba(67, 83, 255, 0.2)',
      drag: false,
      resize: false,
    });
  });
}

wavesurfer.on('ready', () => {
  loadVTTRegions('/path/to/subtitles.vtt');
});

Using this workflow, any compliant WebVTT file automatically generates functional, labeled markers across the timeline as soon as the audio finishes decoding.
