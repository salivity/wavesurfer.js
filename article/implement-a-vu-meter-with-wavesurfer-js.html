Implement a VU Meter with Wavesurfer.js

This guide explains how to build a real-time Volume Unit (VU) meter alongside an interactive Wavesurfer.js waveform. By leveraging the Web Audio API’s AnalyserNode connected directly to the Wavesurfer.js audio source, you can capture real-time audio signal levels, compute Root Mean Square (RMS) or peak volume, and display a responsive visual level meter synced with playback.

1. Set Up the HTML Structure

Create an interface containing containers for both the Wavesurfer waveform and the VU meter.

<div class="audio-player">
  <div id="waveform"></div>
  <div class="vu-meter-container">
    <div id="vu-meter" class="vu-meter-bar"></div>
  </div>
  <button id="play-btn">Play / Pause</button>
</div>

<style>
  .audio-player {
    display: flex;
    flex-direction: column;
    gap: 12px;
    width: 100%;
    max-width: 600px;
  }
  .vu-meter-container {
    width: 100%;
    height: 12px;
    background-color: #e0e0e0;
    border-radius: 6px;
    overflow: hidden;
  }

    width: 0%;
    height: 100%;
    background: linear-gradient(to right, #4caf50 70%, #ffeb3b 85%, #f44336 100%);
    transition: width 50ms ease-out;
  }
</style>

2. Initialize Wavesurfer.js

Initialize Wavesurfer and load your audio file. Make sure Wavesurfer is available via your module bundler or a script tag.

import WaveSurfer from 'wavesurfer.js';

const wavesurfer = WaveSurfer.create({
  container: '#waveform',
  waveColor: '#4F4A85',
  progressColor: '#383358',
  url: 'path/to/your-audio.mp3',
});

const playButton = document.getElementById('play-btn');
playButton.addEventListener('click', () => {
  wavesurfer.playPause();
});

3. Connect to the Web Audio API

To measure audio data dynamically, attach an AnalyserNode to the audio pipeline. Wavesurfer provides access to its underlying HTML5 audio element through the getMediaElement() method.

let audioContext;
let analyser;
let sourceNode;
let dataArray;
let isAudioInitialized = false;

function initAudioContext() {
  if (isAudioInitialized) return;

  const audioElement = wavesurfer.getMediaElement();
  audioContext = new (window.AudioContext || window.webkitAudioContext)();
  
  analyser = audioContext.createAnalyser();
  analyser.fftSize = 256;
  
  sourceNode = audioContext.createMediaElementSource(audioElement);
  sourceNode.connect(analyser);
  analyser.connect(audioContext.destination);

  dataArray = new Uint8Array(analyser.frequencyBinCount);
  isAudioInitialized = true;
}

wavesurfer.on('play', () => {
  initAudioContext();
  if (audioContext.state === 'suspended') {
    audioContext.resume();
  }
  renderVUMeter();
});

4. Calculate Signal Volume and Animate

Use requestAnimationFrame to continuously extract time-domain audio data, compute the Root Mean Square (RMS) value, and update the meter width.

const vuMeterBar = document.getElementById('vu-meter');
let animationFrameId;

function renderVUMeter() {
  if (!wavesurfer.isPlaying()) {
    vuMeterBar.style.width = '0%';
    cancelAnimationFrame(animationFrameId);
    return;
  }

  analyser.getByteTimeDomainData(dataArray);

  // Compute Root Mean Square (RMS) to determine volume
  let sumSquares = 0.0;
  for (let i = 0; i < dataArray.length; i++) {
    // Convert 0-255 range to -1.0 to 1.0 range
    const normalized = (dataArray[i] / 128) - 1.0;
    sumSquares += normalized * normalized;
  }

  const rms = Math.sqrt(sumSquares / dataArray.length);
  
  // Scale RMS to a visual percentage
  let volumePercent = Math.min(Math.round(rms * 200), 100);

  vuMeterBar.style.width = `${volumePercent}%`;

  animationFrameId = requestAnimationFrame(renderVUMeter);
}

wavesurfer.on('pause', () => {
  cancelAnimationFrame(animationFrameId);
  vuMeterBar.style.width = '0%';
});

wavesurfer.on('finish', () => {
  cancelAnimationFrame(animationFrameId);
  vuMeterBar.style.width = '0%';
});

5. Performance and Resource Cleanup

Always ensure cleanup routines are in place when destroying the player instance:

  • Invoke cancelAnimationFrame(animationFrameId) to avoid background visual loops.
  • Call audioContext.close() when removing or navigating away from the audio player to release audio hardware resources.
